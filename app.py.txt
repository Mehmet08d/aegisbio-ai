def ncbi_gen_getir(accession_id, gen_adi):
    try:
        handle = Entrez.efetch(db="nucleotide", id=accession_id, rettype="gb", retmode="text")
        record = SeqIO.read(handle, "genbank")
        handle.close()

        gen_sozlugu = {
            "ND1": ["ND1", "NAD1"], "ND2": ["ND2", "NAD2"], "ND3": ["ND3", "NAD3"],
            "ND4": ["ND4", "NAD4"], "ND4L": ["ND4L", "NAD4L"], "ND5": ["ND5", "NAD5"], "ND6": ["ND6", "NAD6"],
            "CYTB": ["CYTB", "CYTOCHROME B", "COB"], "COX1": ["COX1", "COI", "COXI"],
            "COX2": ["COX2", "COII"], "COX3": ["COX3", "COIII"], "ATP6": ["ATP6", "ATPASE 6"], "ATP8": ["ATP8", "ATPASE 8"]
        }
        hedef_etiketler = gen_sozlugu.get(gen_adi, [gen_adi])

        for feature in record.features:
            if feature.type == "CDS":
                gene_tag = feature.qualifiers.get("gene", [""])[0].upper()
                product_tag = feature.qualifiers.get("product", [""])[0].upper()
                if any(e in gene_tag or e in product_tag for e in hedef_etiketler):
                    if "translation" in feature.qualifiers:
                        return feature.qualifiers["translation"][0]
        return None
    except Exception:
        return None

def dizileri_hizala(seq1, seq2):
    aligner = PairwiseAligner()
    aligner.mode = 'global'
    alignment = aligner.align(seq1, seq2)[0]
    aligned_seqs = str(alignment).splitlines()
    return aligned_seqs[0], aligned_seqs[2]

def gen_analiz_et(gene_code, numbat_acc, thylacine_acc):
    p1 = ncbi_gen_getir(numbat_acc, gene_code)
    p2 = ncbi_gen_getir(thylacine_acc, gene_code)

    if not p1 or not p2:
        return None

    al1, al2 = dizileri_hizala(p1, p2)
    farklar = []
    toplam_risk = 0.0

    for idx in range(min(len(al1), len(al2))):
        aa1, aa2 = al1[idx], al2[idx]
        if aa1 != "-" and aa2 != "-" and aa1 != aa2:
            pr1 = AMINO_ACID_PROPERTIES.get(aa1, {'charge': 0, 'hydrophobicity': 0})
            pr2 = AMINO_ACID_PROPERTIES.get(aa2, {'charge': 0, 'hydrophobicity': 0})

            yuk_f = abs(pr1['charge'] - pr2['charge'])
            hid_f = abs(pr1['hydrophobicity'] - pr2['hydrophobicity'])
            lokal_r = (yuk_f * 2.5) + (hid_f * 1.0)
            toplam_risk += lokal_r

            farklar.append({
                "Gen": gene_code, "Pozisyon": idx + 1, 
                "Tur 1 (AA)": aa1, "Tur 2 (AA)": aa2,
                "Yuk Degisimi": yuk_f, "Hidrofobiklik Degisimi": round(hid_f, 2), 
                "Lokal Risk Skoru": round(lokal_r, 2)
            })

    risk_skoru = min(100.0, (toplam_risk / (max(len(al1), 1) * 5.0)) * 300)
    return {"gene": gene_code, "seq_len": len(al1), "mutations": len(farklar), "risk_score": risk_skoru, "farklar": farklar}

def pdf_rapor_olustur(tur1_ad, tur2_ad, gen_adi, risk_skoru, toplam_mutasyon, detay_df):
    buffer = io.BytesIO()
    doc = SimpleDocTemplate(buffer, pagesize=letter, rightMargin=30, leftMargin=30, topMargin=30, bottomMargin=30)
    story = []
    styles = getSampleStyleSheet()

    title_style = ParagraphStyle('TitleStyle', parent=styles['Heading1'], fontSize=18, textColor=colors.HexColor('#0284c7'), spaceAfter=10)
    story.append(Paragraph("AegisBio AI - Biyoinformatik Analiz Raporu", title_style))
    story.append(Paragraph("<b>Platform:</b> AegisBio AI Mitonuclear Compatibility & De-Extinction Suite", styles['Normal']))
    story.append(Spacer(1, 15))

    summary_data = [
        ["Parametre", "Deger"],
        ["1. Tur (Tasiyici)", tur1_ad],
        ["2. Tur (Hedef)", tur2_ad],
        ["Incelenen Gen", gen_adi],
        ["Toplam Mutasyonlu Lokus", str(toplam_mutasyon)],
        ["Mitonukleer Risk Skoru", f"%{risk_skoru:.2f}"]
    ]
    t_summary = Table(summary_data, colWidths=[200, 300])
    t_summary.setStyle(TableStyle([
        ('BACKGROUND', (0,0), (-1,0), colors.HexColor('#0f172a')),
        ('TEXTCOLOR', (0,0), (-1,0), colors.white),
        ('FONTNAME', (0,0), (-1,0), 'Helvetica-Bold'),
        ('BOTTOMPADDING', (0,0), (-1,-1), 6),
        ('GRID', (0,0), (-1,-1), 1, colors.HexColor('#cbd5e1')),
    ]))
    story.append(t_summary)
    story.append(Spacer(1, 20))

    story.append(Paragraph("<b>Mito-CRISPR Duzenleme Aday Lokuslari (Ilk 15 Nokta)</b>", styles['Heading2']))
    story.append(Spacer(1, 10))

    if not detay_df.empty:
        table_data = [list(detay_df.columns)]
        for idx, row in detay_df.head(15).iterrows():
            table_data.append([str(val) for val in row.values])

        t_detay = Table(table_data, colWidths=[60, 60, 70, 70, 80, 90, 80])
        t_detay.setStyle(TableStyle([
            ('BACKGROUND', (0,0), (-1,0), colors.HexColor('#334155')),
            ('TEXTCOLOR', (0,0), (-1,0), colors.white),
            ('FONTSIZE', (0,0), (-1,-1), 8),
            ('GRID', (0,0), (-1,-1), 0.5, colors.HexColor('#94a3b8')),
        ]))
        story.append(t_detay)

    doc.build(story)
    buffer.seek(0)
    return buffer
